cmake_minimum_required(VERSION 3.16)
project(otello VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE OTELLO_MACOS_SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(EXISTS "${OTELLO_MACOS_SDK}/usr/include/c++/v1")
    include_directories(SYSTEM "${OTELLO_MACOS_SDK}/usr/include/c++/v1")
  endif()
endif()
add_library(othello SHARED cpp/search.cpp cpp/move_policy.cpp cpp/nnue.cpp cpp/pattern.cpp cpp/cnn.cpp cpp/resnet.cpp cpp/mcts.cpp cpp/default_engine.cpp cpp/api.cpp)
target_include_directories(othello PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cpp)
option(OTELLO_PATTERN_SIMD "Integer Pattern SIMD kernels" ON)
if(OTELLO_PATTERN_SIMD)
  target_compile_definitions(othello PUBLIC OTELLO_PATTERN_SIMD=1)
endif()
foreach(OTELLO_PATTERN_OPT INTEGER_ROUNDING FAST_CODES BLOCK4 LEGAL_POLICY)
  set(OTELLO_PATTERN_OPT_DEFAULT ON)
  if(OTELLO_PATTERN_OPT STREQUAL "INTEGER_ROUNDING")
    set(OTELLO_PATTERN_OPT_DEFAULT OFF)
  endif()
  option(OTELLO_PATTERN_${OTELLO_PATTERN_OPT} "Pattern ${OTELLO_PATTERN_OPT}" ${OTELLO_PATTERN_OPT_DEFAULT})
  if(OTELLO_PATTERN_${OTELLO_PATTERN_OPT})
    target_compile_definitions(othello PUBLIC OTELLO_PATTERN_${OTELLO_PATTERN_OPT}=1)
  endif()
endforeach()
option(OTELLO_WITH_EGAROUCID "Enable the native Egaroucid adapter" OFF)
set(OTELLO_EGAROUCID_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.build/deps/Egaroucid" CACHE PATH "Pinned Egaroucid source checkout")
if(OTELLO_WITH_EGAROUCID)
  if(NOT EXISTS "${OTELLO_EGAROUCID_SOURCE}/include/egaroucid/egaroucid.h")
    message(FATAL_ERROR "Run sh scripts/setup-egaroucid.sh first, or set OTELLO_EGAROUCID_SOURCE")
  endif()
  set(BUILD_ENGINE_LIB ON CACHE BOOL "" FORCE)
  set(BUILD_CONSOLE OFF CACHE BOOL "" FORCE)
  set(BUILD_GUI OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(HAS_NO_AVX2 ON CACHE BOOL "Portable Egaroucid build" FORCE)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)$")
    set(HAS_ARM_PROCESSOR ON CACHE BOOL "" FORCE)
  endif()
  add_subdirectory("${OTELLO_EGAROUCID_SOURCE}" "${CMAKE_BINARY_DIR}/egaroucid" EXCLUDE_FROM_ALL)
  get_target_property(OTELLO_EGAROUCID_SOURCES egaroucid SOURCES)
  list(LENGTH OTELLO_EGAROUCID_SOURCES OTELLO_EGAROUCID_SOURCE_COUNT)
  if(NOT OTELLO_EGAROUCID_SOURCE_COUNT EQUAL 1 OR NOT OTELLO_EGAROUCID_SOURCES MATCHES "egaroucid_c_api.cpp$")
    message(FATAL_ERROR "Pinned Egaroucid source layout changed; review the book-only bridge")
  endif()
  set_property(TARGET egaroucid PROPERTY SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/cpp/egaroucid_bridge.cpp")
  set_target_properties(egaroucid PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_sources(othello PRIVATE cpp/egaroucid.cpp)
  target_link_libraries(othello PRIVATE egaroucid)
  target_compile_definitions(othello PUBLIC OTELLO_WITH_EGAROUCID=1)
  target_compile_definitions(othello PRIVATE OTELLO_EGAROUCID_RESOURCES="${OTELLO_EGAROUCID_SOURCE}/bin/resources")
  add_executable(otello_egaroucid_tests tests/egaroucid.cpp)
  target_link_libraries(otello_egaroucid_tests PRIVATE othello egaroucid)
endif()
add_executable(otello cpp/main.cpp)
target_link_libraries(otello PRIVATE othello)
add_executable(otello_dataset cpp/dataset.cpp)
target_link_libraries(otello_dataset PRIVATE othello)
add_executable(otello_tests tests/native.cpp)
target_link_libraries(otello_tests PRIVATE othello)
add_executable(otello_pattern_benchmark benchmarks/pattern-hotpath.cpp)
target_link_libraries(otello_pattern_benchmark PRIVATE othello)
add_executable(otello_pattern_legal_benchmark benchmarks/pattern-legal.cpp)
target_link_libraries(otello_pattern_legal_benchmark PRIVATE othello)
enable_testing()
add_test(NAME rules_and_search COMMAND otello_tests)
if(OTELLO_WITH_EGAROUCID)
  add_test(NAME egaroucid_integration COMMAND otello_egaroucid_tests)
  set_tests_properties(egaroucid_integration PROPERTIES TIMEOUT 120)
endif()
install(TARGETS otello othello)
install(FILES cpp/api.h cpp/othello.hpp cpp/nnue.hpp cpp/pattern.hpp cpp/cnn.hpp cpp/resnet.hpp cpp/mcts.hpp cpp/default_engine.hpp DESTINATION include/otello)
if(OTELLO_WITH_EGAROUCID)
  install(FILES cpp/egaroucid.hpp cpp/egaroucid_api.h DESTINATION include/otello)
endif()

add_executable(otello_global_policy_tests tests/global_policy_native.cpp)
target_link_libraries(otello_global_policy_tests PRIVATE othello)
add_test(NAME global_policy_native COMMAND otello_global_policy_tests "${CMAKE_CURRENT_SOURCE_DIR}/models/pattern32-global-policy-v2.otpat")
